# Root quadlet on the core host: one pod holding core and mongo.
# Install under /etc/containers/systemd/quomodo/ with the two .container
# files and the .volume files. See docs/install.md.
#
# Mongo publishes nothing. Core publishes 9120, which the UI, the API and
# the agents all use. Agents only connect over wss://: either put a TLS
# reverse proxy in front and bind this to an address only the proxy reaches
# (PublishPort=127.0.0.1:9120:9120), or turn on TLS in quomodo-core.container.
# If the address comes from a VPN daemon, order the pod after it so the
# address exists at boot:
#   After=tailscaled.service
#   Wants=tailscaled.service
[Unit]
Description=Quomodo core pod
After=network-online.target
Wants=network-online.target

[Pod]
PodName=quomodo
PublishPort=9120:9120

[Install]
WantedBy=multi-user.target
