# Root quadlet on the core host: one pod holding core and mongo.
# Install under /etc/containers/systemd/quomodo/ with the two .container files.
#
# Mongo publishes nothing. Core publishes only on the tailnet address,
# and the pod waits for tailscaled so that address exists at boot.
[Unit]
Description=Quomodo core pod
After=tailscaled.service network-online.target
Wants=tailscaled.service network-online.target

[Pod]
PodName=quomodo
PublishPort=10.0.0.5:9120:9120

[Install]
WantedBy=multi-user.target
